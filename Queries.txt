CREATE TABLE Sales.FactSales
(
    SaleID BIGINT IDENTITY(1,1) PRIMARY KEY,
    DateID INT NOT NULL,
    CustomerID INT NOT NULL,
    ProductID INT NOT NULL,
    RegionID INT NOT NULL,
    EntityID INT NOT NULL,
    SalesRepEmployeeID INT NOT NULL,
    VersionID INT NOT NULL,
    Quantity INT NOT NULL,
    UnitPrice DECIMAL(18,2) NOT NULL,
    Revenue DECIMAL(18,2) NOT NULL,
    Cost DECIMAL(18,2) NOT NULL,
    Margin AS (Revenue - Cost) PERSISTED,

    CONSTRAINT FK_FactSales_Date FOREIGN KEY (DateID) REFERENCES MasterData.DimDate(DateID),
    CONSTRAINT FK_FactSales_Customer FOREIGN KEY (CustomerID) REFERENCES MasterData.DimCustomer(CustomerID),
    CONSTRAINT FK_FactSales_Product FOREIGN KEY (ProductID) REFERENCES MasterData.DimProduct(ProductID),
    CONSTRAINT FK_FactSales_Region FOREIGN KEY (RegionID) REFERENCES MasterData.DimRegion(RegionID),
    CONSTRAINT FK_FactSales_Entity FOREIGN KEY (EntityID) REFERENCES MasterData.DimEntity(EntityID),
    CONSTRAINT FK_FactSales_Employee FOREIGN KEY (SalesRepEmployeeID) REFERENCES MasterData.DimEmployee(EmployeeID),
    CONSTRAINT FK_FactSales_Version FOREIGN KEY (VersionID) REFERENCES MasterData.DimVersion(VersionID)
);
GO

CREATE OR ALTER VIEW Sales.vw_RevenueCube_Source
AS
SELECT
    D.YearNumber,
    D.QuarterName,
    D.MonthName,
    FS.DateID,

    C.CustomerID,
    C.CustomerCode,
    C.CustomerName,
    C.Industry,
    C.CustomerSegment,
    C.RevenueBand,

    P.ProductID,
    P.ProductCode,
    P.ProductName,
    P.ProductCategory,
    P.ProductFamily,
    P.ProductLine,

    R.RegionID,
    R.RegionCode,
    R.RegionName,

    E.EntityID,
    E.EntityCode,
    E.EntityName,

    V.VersionID,
    V.VersionCode,
    V.VersionName,

    FS.Quantity,
    FS.Revenue,
    FS.Cost,
    (FS.Revenue - FS.Cost) AS Margin,
    CASE
        WHEN FS.Revenue = 0 THEN 0
        ELSE (FS.Revenue - FS.Cost) / FS.Revenue
    END AS MarginPercent,
    CASE
        WHEN FS.Quantity = 0 THEN 0
        ELSE FS.Revenue / FS.Quantity
    END AS AverageSellingPrice
FROM Sales.FactSales FS
INNER JOIN MasterData.DimDate D
    ON FS.DateID = D.DateID
INNER JOIN MasterData.DimCustomer C
    ON FS.CustomerID = C.CustomerID
INNER JOIN MasterData.DimProduct P
    ON FS.ProductID = P.ProductID
INNER JOIN MasterData.DimRegion R
    ON FS.RegionID = R.RegionID
INNER JOIN MasterData.DimEntity E
    ON FS.EntityID = E.EntityID
INNER JOIN MasterData.DimVersion V
    ON FS.VersionID = V.VersionID;
GO
